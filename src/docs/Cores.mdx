import tokens from '../styles/tokens.json';

<Meta title="Fundamentos/Cores" />

export const group = (prefix) =>
  Object.fromEntries(
    Object.entries(tokens).filter(([name]) => name.startsWith(prefix)).map(([name]) => [
      name.replace(prefix, ''),
      `var(--${name})`,
    ])
  );
